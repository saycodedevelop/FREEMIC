cmake_minimum_required(VERSION 3.22.1)
project(karaokelink LANGUAGES CXX)
find_package(oboe REQUIRED CONFIG)
add_library(karaokelink SHARED Engine.cpp)
target_compile_features(karaokelink PRIVATE cxx_std_17)
target_compile_options(karaokelink PRIVATE -Wall -Wextra -Werror=return-type)
target_link_options(karaokelink PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(karaokelink PRIVATE oboe::oboe android log)
